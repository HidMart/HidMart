نصب

برای نصب HidMart از pip استفاده کنید:

pip install hidmart

شروع سریع

یک Bot ساده با دستور start:

from hidmart import Bot

bot = Bot("TOKEN")

@bot.on_command("start")
async def start(message):
    await message.reply("سلام!")

bot.run()

Bot

ساخت Bot

bot = Bot("TOKEN")

تنظیمات Bot

bot = Bot(
    "TOKEN",
    poll_interval=1.0,
    timeout=25
)

Handler

دریافت Command

@bot.on_command("start")
async def start(message):
    await message.reply("سلام!")

چند Command

@bot.on_command(
    "start",
    "help"
)
async def commands(message):
    await message.reply("دستور دریافت شد")

دریافت پیام

@bot.on_message()
async def message_handler(message):
    print(message.text)

دریافت Text مشخص

@bot.on_text("سلام")
async def hello(message):
    await message.reply("سلام!")

Message

اطلاعات پیام

message.message_id
message.id
message.text
message.chat
message.from_user
message.sender
message.raw

پاسخ به پیام

await message.reply("سلام!")

پاسخ با answer

await message.answer("سلام!")

حذف پیام

await message.delete()

User

اطلاعات کاربر از طریق from_user قابل دسترسی است.

user = message.from_user

user.id
user.first_name
user.last_name
user.username
user.full_name

Chat

اطلاعات گفتگو از طریق message.chat قابل دسترسی است.

chat = message.chat

chat.id
chat.type
chat.title
chat.username

بررسی Private

if message.chat.is_private:
    print("Private")

بررسی Group

if message.chat.is_group:
    print("Group")

اطلاعات Bot و Chat

اطلاعات Bot

اطلاعات Chat

    chat_id
)

اطلاعات عضو

)

Updates

دریافت Updates

تنظیمات Updates

updates = await bot.get_updates(
    offset=10,
    timeout=25,
    limit=10
)

bot.polling()

مدیریت خطاها

خطاهای اصلی HidMart:

  • HidMartError
  • APIError
  • NetworkError
  • InvalidTokenError

استفاده

from hidmart import (
    HidMartError,
    APIError,
    NetworkError,
    InvalidTokenError
)

مثال

try:
    await bot.send_message(
        chat_id,
        "Hello"
    )

except InvalidTokenError:
    print("Invalid Token")
