نصب

برای نصب آخرین نسخه HidMart از دستور زیر استفاده کنید:

pip install hidmart

شروع سریع

from hidmart import Bot

bot = Bot("TOKEN")

@bot.on_command("start")
async def start(message):
    await message.reply(
        "سلام 👋"
    )

bot.run()

Bot

ساخت Bot

bot = Bot("TOKEN")

تنظیمات Bot

bot = Bot(
    "TOKEN",
    poll_interval=1.0,
    timeout=25
)

Handlerها

Command Handler

@bot.on_command("start")
async def start(message):
    await message.reply("سلام!")

چند Command

@bot.on_command("start", "help")
async def commands(message):
    await message.reply(
        "دستور دریافت شد"
    )

دریافت پیام

@bot.on_message()
async def message_handler(message):
    print(message.text)

دریافت Text مشخص

@bot.on_text("سلام")
async def hello(message):
    await message.reply(
        "سلام 👋"
    )

مدیریت خطا

خطاهای موجود

  • HidMartError
  • APIError
  • NetworkError
  • InvalidTokenError

Import خطاها

    HidMartError,
    APIError,
    NetworkError,
    InvalidTokenError
)

مثال

try:
    await bot.send_message(
        chat_id,
        "Hello"
    )

except InvalidTokenError:
    print("Invalid Token")
